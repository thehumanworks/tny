import hashlib,json,os,subprocess,sys,time
from pathlib import Path
root=Path.cwd()
env=os.environ.copy()
for key in list(env):
    if key == 'TNY_TOOLS' or key.endswith(('_BASE_URL','_API_KEY','_API_KEY_CMD','_WIRE_API')):
        del env[key]
env['TMPDIR']=str(root/'build/tmp')
env['CLANG_MODULE_CACHE_PATH']=str(root/'build/module-cache')
env['SWIFT_MODULECACHE_PATH']=str(root/'build/module-cache')
env['PATH']=str(root/'build/review2/bin')+os.pathsep+env['PATH']
env['TNY']=str(root/'build/tny')
label=sys.argv[1]
command=sys.argv[2:]
if label == "delivered-quality":
    command = ["make", "-j8", "quality"]
paths=subprocess.check_output(['git','ls-files','--cached','--others','--exclude-standard','src','include','tests','sdk','Makefile','nix','.github'],text=True).splitlines()
manifest={p:hashlib.sha256(Path(p).read_bytes()).hexdigest() for p in sorted(set(paths)) if Path(p).is_file()}
manifest_path=root/'build/review3'/f'{label}-source.json'
manifest_path.write_text(json.dumps(manifest,sort_keys=True,indent=2)+'\n')
start=time.time()
with (root/'build/review3'/f'{label}.log').open('w') as out:
    result=subprocess.run(command,env=env,stdout=out,stderr=subprocess.STDOUT)
record={'label':label,'command':command,'exit':result.returncode,'seconds':time.time()-start,'source_manifest_sha256':hashlib.sha256(manifest_path.read_bytes()).hexdigest(),'revision':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()}
with (root/'build/review3/runs.jsonl').open('a') as out: out.write(json.dumps(record)+'\n')
print(json.dumps(record))
print((root/'build/review3'/f'{label}.log').read_text()[-2000:])
sys.exit(result.returncode)
